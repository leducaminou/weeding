components\admin\EventsManager.tsx
"use client";

import { useState } from "react";
import { useRouter } from "next/navigation";
import { Plus, Pencil, Trash2, Loader2, GripVertical } from "lucide-react";
import { formatDateFR } from "@/lib/utils";
import { AdminCard } from "./AdminCard";
import { EventFormDialog } from "./EventFormDialog";

interface Event {
  id: string;
  title: string;
  description: string | null;
  date: Date;
  time: string;
  location: string;
  address: string | null;
  mapsUrl: string | null;
  position: number;
  active: boolean;
}

export function EventsManager({ initialItems }: { initialItems: Event[] }) {
  const router = useRouter();
  const [editing, setEditing] = useState<Event | null>(null);
  const [creating, setCreating] = useState(false);
  const [busy, setBusy] = useState<string | null>(null);

  async function remove(id: string, title: string) {
    if (!confirm(`Supprimer "${title}" ?`)) return;
    setBusy(id);
    try {
      await fetch(`/api/admin/events/${id}`, { method: "DELETE" });
      router.refresh();
    } finally {
      setBusy(null);
    }
  }

  return (
    <>
      <div className="flex justify-end mb-4">
        <button
          onClick={() => setCreating(true)}
          className="inline-flex items-center gap-2 h-10 px-4 bg-slate-900 text-white text-sm rounded-md hover:bg-slate-800"
        >
          <Plus size={14} />
          Nouvel événement
        </button>
      </div>

      {initialItems.length === 0 ? (
        <AdminCard className="p-10 text-center text-sm text-slate-400">
          Aucun événement.
        </AdminCard>
      ) : (
        <div className="space-y-3">
          {initialItems.map((item) => (
            <AdminCard key={item.id} className="p-4 flex items-start gap-4">
              <GripVertical size={16} className="text-slate-300 mt-1 shrink-0" />
              <div className="flex-1 min-w-0">
                <div className="flex items-center gap-2 flex-wrap">
                  <h3 className="font-medium text-slate-900">{item.title}</h3>
                  {!item.active && (
                    <span className="text-[10px] uppercase px-2 py-0.5 rounded-full bg-slate-100 text-slate-500">
                      Inactif
                    </span>
                  )}
                </div>
                <p className="mt-1 text-xs text-slate-500">
                  {formatDateFR(item.date)} — {item.time} — {item.location}
                </p>
                {item.description && (
                  <p className="mt-2 text-sm text-slate-600 line-clamp-2">
                    {item.description}
                  </p>
                )}
              </div>
              <div className="flex items-center gap-1 shrink-0">
                {busy === item.id ? (
                  <Loader2 size={14} className="animate-spin text-slate-400" />
                ) : (
                  <>
                    <button
                      onClick={() => setEditing(item)}
                      className="p-1.5 rounded hover:bg-slate-100 text-slate-500 hover:text-slate-900"
                    >
                      <Pencil size={14} />
                    </button>
                    <button
                      onClick={() => remove(item.id, item.title)}
                      className="p-1.5 rounded hover:bg-red-50 text-slate-500 hover:text-red-600"
                    >
                      <Trash2 size={14} />
                    </button>
                  </>
                )}
              </div>
            </AdminCard>
          ))}
        </div>
      )}

      <EventFormDialog
        open={creating}
        onClose={() => setCreating(false)}
        onSaved={() => {
          setCreating(false);
          router.refresh();
        }}
      />
      <EventFormDialog
        open={!!editing}
        event={editing ?? undefined}
        onClose={() => setEditing(null)}
        onSaved={() => {
          setEditing(null);
          router.refresh();
        }}
      />
    </>
  );
}

components\admin\EventFormDialog.tsx
"use client";

import { useState } from "react";
import { Loader2 } from "lucide-react";
import { toDate } from "@/lib/date";

interface Event {
  id: string;
  title: string;
  description: string | null;
  date: string | Date;
  time: string;
  endTime: string | null;
  location: string;
  address: string | null;
  mapsUrl: string | null;
  position: number;
  active: boolean;
  dressCode: string | null;
  dressCodeNote: string | null;
  photoSession: string | null;
}

export function EventFormDialog({
  open,
  event,
  onClose,
  onSaved,
}: {
  open: boolean;
  event?: Event;
  onClose: () => void;
  onSaved: () => void;
}) {
  const [submitting, setSubmitting] = useState(false);
  const [error, setError] = useState<string | null>(null);
  if (!open) return null;

  async function onSubmit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    setSubmitting(true);
    setError(null);
    const fd = new FormData(e.currentTarget);
    const payload = {
      title: String(fd.get("title") ?? ""),
      description: String(fd.get("description") ?? "") || null,
      date: new Date(String(fd.get("date") ?? "")).toISOString(),
      time: String(fd.get("time") ?? ""),
      endTime: String(fd.get("endTime") ?? "") || null,
      location: String(fd.get("location") ?? ""),
      address: String(fd.get("address") ?? "") || null,
      mapsUrl: String(fd.get("mapsUrl") ?? "") || null,
      position: Number(fd.get("position") ?? 0),
      active: fd.get("active") === "on",
      dressCode: String(fd.get("dressCode") ?? "") || null,
      dressCodeNote: String(fd.get("dressCodeNote") ?? "") || null,
      photoSession: String(fd.get("photoSession") ?? "") || null,
    };
    try {
      const url = event
        ? `/api/admin/events/${event.id}`
        : "/api/admin/events";
      const method = event ? "PATCH" : "POST";
      const res = await fetch(url, {
        method,
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(payload),
      });
      const data = await res.json();
      if (!res.ok) {
        setError(data.error ?? "Erreur");
        setSubmitting(false);
        return;
      }
      onSaved();
    } catch {
      setError("Erreur serveur");
      setSubmitting(false);
    }
  }

  const inputCls =
    "mt-1 w-full px-3 py-2 text-sm bg-white border border-slate-200 rounded-md focus:outline-none focus:ring-2 focus:ring-slate-900/10 focus:border-slate-400";
  const dateValue = event
    ? toDate(event.date).toISOString().slice(0, 10)
    : "";

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/50">
      <div className="w-full max-w-lg bg-white rounded-lg shadow-xl max-h-[90vh] overflow-y-auto">
        <form onSubmit={onSubmit} className="p-6">
          <h2 className="text-lg font-semibold text-slate-900 mb-5">
            {event ? "Modifier l'événement" : "Nouvel événement"}
          </h2>
          <div className="space-y-4">
            <div>
              <label className="text-xs text-slate-500 uppercase tracking-wide">
                Titre *
              </label>
              <input
                name="title"
                defaultValue={event?.title}
                required
                className={inputCls}
              />
            </div>
            <div>
              <label className="text-xs text-slate-500 uppercase tracking-wide">
                Description
              </label>
              <textarea
                name="description"
                defaultValue={event?.description ?? ""}
                rows={3}
                className={inputCls}
              />
            </div>
            <div className="grid grid-cols-3 gap-3">
              <div>
                <label className="text-xs text-slate-500 uppercase tracking-wide">
                  Date *
                </label>
                <input
                  name="date"
                  type="date"
                  defaultValue={dateValue}
                  required
                  className={inputCls}
                />
              </div>
              <div>
                <label className="text-xs text-slate-500 uppercase tracking-wide">
                  Heure début *
                </label>
                <input
                  name="time"
                  defaultValue={event?.time}
                  placeholder="10h00"
                  required
                  className={inputCls}
                />
              </div>
              <div>
                <label className="text-xs text-slate-500 uppercase tracking-wide">
                  Heure fin
                </label>
                <input
                  name="endTime"
                  defaultValue={event?.endTime ?? ""}
                  placeholder="18h00"
                  className={inputCls}
                />
              </div>
            </div>
            <div>
              <label className="text-xs text-slate-500 uppercase tracking-wide">
                Lieu *
              </label>
              <input
                name="location"
                defaultValue={event?.location}
                required
                className={inputCls}
              />
            </div>
            <div>
              <label className="text-xs text-slate-500 uppercase tracking-wide">
                Adresse
              </label>
              <input
                name="address"
                defaultValue={event?.address ?? ""}
                className={inputCls}
              />
            </div>
            <div>
              <label className="text-xs text-slate-500 uppercase tracking-wide">
                Lien Google Maps
              </label>
              <input
                name="mapsUrl"
                type="url"
                defaultValue={event?.mapsUrl ?? ""}
                className={inputCls}
              />
            </div>

            {/* Champs dress code */}
            <div className="pt-3 border-t border-slate-100">
              <p className="text-xs text-slate-500 uppercase tracking-wide mb-2">
                Dress code (optionnel — uniquement pour certains événements)
              </p>
              <div className="space-y-3">
                <div>
                  <label className="text-xs text-slate-500">Dress code</label>
                  <input
                    name="dressCode"
                    defaultValue={event?.dressCode ?? ""}
                    placeholder="Ex : Bleu & Blanc"
                    className={inputCls}
                  />
                </div>
                <div>
                  <label className="text-xs text-slate-500">Note dress code</label>
                  <textarea
                    name="dressCodeNote"
                    defaultValue={event?.dressCodeNote ?? ""}
                    rows={2}
                    placeholder="Explication du dress code"
                    className={inputCls}
                  />
                </div>
              </div>
            </div>

            {/* Séance photo */}
            <div>
              <label className="text-xs text-slate-500 uppercase tracking-wide">
                Séance photo (optionnel)
              </label>
              <input
                name="photoSession"
                defaultValue={event?.photoSession ?? ""}
                placeholder="Ex : 1h de séance photo"
                className={inputCls}
              />
            </div>

            <div className="grid grid-cols-2 gap-3">
              <div>
                <label className="text-xs text-slate-500 uppercase tracking-wide">
                  Position
                </label>
                <input
                  name="position"
                  type="number"
                  defaultValue={String(event?.position ?? 0)}
                  className={inputCls}
                />
              </div>
              <div className="flex items-center gap-2 mt-6">
                <input
                  type="checkbox"
                  id="active"
                  name="active"
                  defaultChecked={event?.active ?? true}
                  className="h-4 w-4"
                />
                <label htmlFor="active" className="text-sm text-slate-700">
                  Actif
                </label>
              </div>
            </div>
          </div>
          {error && (
            <p className="mt-4 text-xs text-red-600 text-center">{error}</p>
          )}
          <div className="flex justify-end gap-2 mt-6">
            <button
              type="button"
              onClick={onClose}
              className="h-10 px-4 text-sm text-slate-600 hover:text-slate-900"
            >
              Annuler
            </button>
            <button
              type="submit"
              disabled={submitting}
              className="h-10 px-4 bg-slate-900 text-white text-sm rounded-md hover:bg-slate-800 disabled:opacity-50 inline-flex items-center gap-2"
            >
              {submitting && <Loader2 size={14} className="animate-spin" />}
              Enregistrer
            </button>
          </div>
        </form>
      </div>
    </div>
  );
}




model Event {
  id          String   @id @default(cuid())
  weddingId   String
  title       String
  description String?
  date        DateTime
  time        String
  endTime    String?
  location    String
  address     String?
  photoSession String?
  latitude    Float?
  longitude   Float?
  mapsUrl     String?
  position    Int      @default(0)
  dressCode      String?
  dressCodeNote  String?
  active      Boolean  @default(true)
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt

  wedding Wedding @relation(fields: [weddingId], references: [id], onDelete: Cascade)

  @@index([weddingId, position])
  @@map("events")
}
